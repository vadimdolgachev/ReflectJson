# Keep executable settings local to the test project.
set(ReflectJson_RUNTIME_DIR "")
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=libstdc++.so
    OUTPUT_VARIABLE ReflectJson_runtime OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(IS_ABSOLUTE "${ReflectJson_runtime}" AND EXISTS "${ReflectJson_runtime}")
    get_filename_component(ReflectJson_runtime "${ReflectJson_runtime}" REALPATH)
    get_filename_component(ReflectJson_RUNTIME_DIR "${ReflectJson_runtime}" DIRECTORY)
endif()

function(ReflectJson_configure_executable target)
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(ReflectJson_RUNTIME_DIR)
        set_property(TARGET ${target} APPEND PROPERTY BUILD_RPATH "${ReflectJson_RUNTIME_DIR}")
    endif()
endfunction()

foreach(case IN ITEMS pointer union private_field private_base map variant nonclass copy move)
    add_executable(ReflectJson_fail_${case} EXCLUDE_FROM_ALL "compile_fail/${case}.cpp")
    target_link_libraries(ReflectJson_fail_${case} PRIVATE ReflectJson::ReflectJson)
    ReflectJson_configure_executable(ReflectJson_fail_${case})
    if(case MATCHES "^(copy|move)$")
        set(marker "deleted")
    elseif(case STREQUAL "nonclass")
        set(marker "MetaClass requires a class type")
    elseif(case MATCHES "^(private_base|variant)$")
        set(marker "MetaClass requires all base classes to be public")
    elseif(case MATCHES "^(private_field|map)$")
        set(marker "MetaClass requires all data members to be public")
    else()
        set(marker "JsonWriter: unsupported type")
    endif()
    add_test(NAME ReflectJson.reject_${case} COMMAND "${CMAKE_COMMAND}"
        "-DBUILD_DIR=${PROJECT_BINARY_DIR}" "-DTARGET=ReflectJson_fail_${case}"
        "-DCONFIG=$<CONFIG>" "-DMARKER=${marker}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check_compile_failure.cmake")
    set_tests_properties(ReflectJson.reject_${case} PROPERTIES RUN_SERIAL TRUE)
endforeach()
foreach(case IN ITEMS string_view pointer union reference_root const_root const_field reference_field bitfield nondefault immovable legacy_name)
    add_executable(ReflectJson_fail_reader_${case} EXCLUDE_FROM_ALL "compile_fail/reader_${case}.cpp")
    target_link_libraries(ReflectJson_fail_reader_${case} PRIVATE ReflectJson::ReflectJson)
    ReflectJson_configure_executable(ReflectJson_fail_reader_${case})
    if(case STREQUAL "legacy_name")
        set(marker "is not a member of")
    elseif(case STREQUAL "string_view")
        set(marker "JsonReader does not support std::string_view targets")
    elseif(case MATCHES "^(reference_root|const_root)$")
        set(marker "JsonReader requires a mutable value target")
    elseif(case STREQUAL "const_field")
        set(marker "MetaClass mutable traversal does not support const members")
    elseif(case STREQUAL "reference_field")
        set(marker "MetaClass mutable traversal does not support reference members")
    elseif(case STREQUAL "bitfield")
        set(marker "MetaClass mutable traversal does not support bitfields")
    elseif(case STREQUAL "nondefault")
        set(marker "JsonReader requires a default-constructible target")
    elseif(case STREQUAL "immovable")
        set(marker "JsonReader requires a copyable or movable target")
    else()
        set(marker "JsonReader: unsupported type")
    endif()
    add_test(NAME ReflectJson.reject_reader_${case} COMMAND "${CMAKE_COMMAND}"
        "-DBUILD_DIR=${PROJECT_BINARY_DIR}" "-DTARGET=ReflectJson_fail_reader_${case}"
        "-DCONFIG=$<CONFIG>" "-DMARKER=${marker}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check_compile_failure.cmake")
    set_tests_properties(ReflectJson.reject_reader_${case} PROPERTIES RUN_SERIAL TRUE)
endforeach()
foreach(case IN ITEMS protected_base ancestor_private_base base_private_field base_protected_field
        hidden_field sibling_name diamond mixed_subobjects private_virtual_path read_const read_ref read_bit)
    add_executable(ReflectJson_fail_inheritance_${case} EXCLUDE_FROM_ALL compile_fail/inheritance.cpp)
    target_compile_definitions(ReflectJson_fail_inheritance_${case} PRIVATE "ReflectJson_CASE_${case}")
    target_link_libraries(ReflectJson_fail_inheritance_${case} PRIVATE ReflectJson::ReflectJson)
    ReflectJson_configure_executable(ReflectJson_fail_inheritance_${case})
    if(case MATCHES "^(protected_base|ancestor_private_base|private_virtual_path)$")
        set(marker "MetaClass requires all base classes to be public")
    elseif(case MATCHES "^base_(private|protected)_field$")
        set(marker "MetaClass requires all data members to be public")
    elseif(case STREQUAL "read_const")
        set(marker "MetaClass mutable traversal does not support const members")
    elseif(case STREQUAL "read_ref")
        set(marker "MetaClass mutable traversal does not support reference members")
    elseif(case STREQUAL "read_bit")
        set(marker "MetaClass mutable traversal does not support bitfields")
    else()
        set(marker "MetaClass requires unique data member names")
    endif()
    add_test(NAME ReflectJson.reject_inheritance_${case} COMMAND "${CMAKE_COMMAND}"
        "-DBUILD_DIR=${PROJECT_BINARY_DIR}" "-DTARGET=ReflectJson_fail_inheritance_${case}"
        "-DCONFIG=$<CONFIG>" "-DMARKER=${marker}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check_compile_failure.cmake")
    set_tests_properties(ReflectJson.reject_inheritance_${case} PROPERTIES RUN_SERIAL TRUE)
endforeach()
