cmake_minimum_required(VERSION 3.30)
project(ReflectJson_consumer LANGUAGES CXX)
if(DEFINED ReflectJson_SOURCE)
    include(CTest)
    add_subdirectory("${ReflectJson_SOURCE}" ReflectJson)
    if(TARGET example OR TARGET ReflectJson_tests_strict OR TARGET ReflectJson_tests_gnu)
        message(FATAL_ERROR "Embedded library created its own example or tests")
    endif()
else()
    find_package(ReflectJson CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp peer.cpp)
target_link_libraries(consumer PRIVATE ReflectJson::ReflectJson)
